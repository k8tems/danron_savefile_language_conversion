from typing import Union

class FileReader:
    def __init__(self, data: Union[bytes, bytearray], endian: str = "little"):
        self.data = data
        self.p = 0
        self.endian = endian

    def tell(self) -> int:
        return self.p

    def seek(self, offset: int):
        self.p = offset

    def skip(self, n: int):
        self.p += n

    def _read(self, n: int) -> bytes:
        chunk = self.data[self.p:self.p + n]
        if len(chunk) != n:
            raise EOFError("Attempted to read past end of buffer")
        self.p += n
        return chunk

    def read_u32(self) -> int:
        return int.from_bytes(self._read(4), self.endian)

    def read_bytes(self, n: int) -> bytes:
        return self._read(n)

    def read_fixed_bytes(self) -> bytes:
        return self._read(self.read_u32())

    def read_str(self, encoding: str = "utf-8") -> str:
        return self.read_fixed_bytes().decode(encoding)
        length = self.read_u32()
        raw = self._read(length)
        return raw.decode(encoding)

    # Optional: peek without advancing cursor
    def peek_u32(self) -> int:
        return int.from_bytes(self.data[self.p:self.p + 4], self.endian)
